cmake_minimum_required(VERSION 3.24)
project(yolo_operator LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

if(CMAKE_SYSTEM_PROCESSOR MATCHES "riscv64")
  set(YOLOOP_X86_DEFAULT OFF)
  set(YOLOOP_RISCV_DEFAULT ON)
else()
  set(YOLOOP_X86_DEFAULT ON)
  set(YOLOOP_RISCV_DEFAULT OFF)
endif()

option(YOLOOP_BUILD_X86         "7840HS/x86 独立 C++ 算子" ${YOLOOP_X86_DEFAULT})
option(YOLOOP_BUILD_RISCV       "RISC-V RVV 1.0 算子（含 K3 X100）" ${YOLOOP_RISCV_DEFAULT})
option(YOLOOP_RISCV_A100 "Run all RISC-V postprocessing on a dedicated K3 A100 core" OFF)
option(YOLOOP_BUILD_CUDA        "NVIDIA CUDA 算子"         ON)
option(YOLOOP_BUILD_TRT_PLUGIN  "TensorRT decode 插件"     ON)
option(YOLOOP_BUILD_PYTHON      "pybind11 Python 绑定"     ON)
option(YOLOOP_BUILD_VULKAN      "Vulkan 混合式 NMS(可选)"  ON)
option(YOLOOP_BUILD_APP         "实时流水线应用 yolo_rt(需 ncnn)" ON)
option(YOLOOP_NCNN_VULKAN       "app 的 ncnn 启用 Vulkan" ON)
option(YOLOOP_BUILD_SPACEMIT_ORT "启用 SpaceMIT ONNX Runtime 后端（自动探测）" ON)
set(SPACEMIT_ORT_ROOT "" CACHE PATH "SpaceMIT ONNX Runtime SDK 根目录，可为空")
set(NCNN_GIT_TAG "20250503" CACHE STRING "FetchContent 拉取的 ncnn release tag")
# 目标机为 Zen4(7840HS) 时: -DYOLOOP_X86_ARCH=znver4
set(YOLOOP_X86_ARCH "native" CACHE STRING "x86 -march 目标")
set(YOLOOP_RISCV_ARCH "rv64gcv" CACHE STRING "RISC-V -march 目标，默认 RVV 1.0")
set(YOLOOP_RISCV_ABI "lp64d" CACHE STRING "RISC-V -mabi 目标")

find_package(OpenMP)

add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-Wall>)

# 公共标量参考实现 + 类型定义
add_library(yoloop_common STATIC common/yolo_ref.cpp)
target_include_directories(yoloop_common PUBLIC common)

if(YOLOOP_BUILD_X86)
  add_subdirectory(x86)
endif()

if(YOLOOP_RISCV_A100 AND NOT YOLOOP_BUILD_RISCV)
  message(FATAL_ERROR "YOLOOP_RISCV_A100 requires YOLOOP_BUILD_RISCV")
endif()

if(YOLOOP_BUILD_RISCV)
  add_subdirectory(riscv)
endif()

if(YOLOOP_BUILD_CUDA)
  include(CheckLanguage)
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    enable_language(CUDA)
    set(CMAKE_CUDA_STANDARD 17)
    if(NOT CMAKE_CUDA_ARCHITECTURES)
      set(CMAKE_CUDA_ARCHITECTURES native)
    endif()
    add_subdirectory(nvidia)
  else()
    message(STATUS "未找到 CUDA 编译器, 跳过 nvidia 模块")
  endif()
endif()

if(YOLOOP_BUILD_VULKAN)
  find_program(GLSLANG_VALIDATOR glslangValidator)
  find_package(Vulkan QUIET)
  if(GLSLANG_VALIDATOR AND Vulkan_FOUND)
    add_subdirectory(vulkan)
  else()
    message(STATUS "未找到 glslangValidator/Vulkan SDK, 跳过 vulkan 混合 NMS "
                   "(在目标机安装 glslang-tools 后自动启用)")
  endif()
endif()

if(YOLOOP_BUILD_APP)
  if(YOLOOP_BUILD_RISCV)
    set(YOLOOP_CPU_LIBRARY yoloop_riscv)
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "riscv64")
      set(YOLOOP_CPU_ARCH_OPTIONS
          -march=${YOLOOP_RISCV_ARCH} -mabi=${YOLOOP_RISCV_ABI})
    endif()
  elseif(YOLOOP_BUILD_X86)
    set(YOLOOP_CPU_LIBRARY yoloop_x86)
    set(YOLOOP_CPU_ARCH_OPTIONS -march=${YOLOOP_X86_ARCH})
  else()
    message(FATAL_ERROR "YOLOOP_BUILD_APP 需要 x86 或 RISC-V CPU 后端")
  endif()

  include(FetchContent)
  set(NCNN_VULKAN ${YOLOOP_NCNN_VULKAN} CACHE BOOL "" FORCE)
  set(NCNN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(NCNN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(NCNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(NCNN_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
  set(NCNN_SHARED_LIB OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(ncnn
    GIT_REPOSITORY https://github.com/Tencent/ncnn.git
    GIT_TAG        ${NCNN_GIT_TAG}
    GIT_SHALLOW    TRUE)
  FetchContent_MakeAvailable(ncnn)
  add_subdirectory(app)
endif()

message(STATUS "==================== yolo_operator 配置 ====================")
message(STATUS "  x86 模块:    ${YOLOOP_BUILD_X86} (march=${YOLOOP_X86_ARCH}, OpenMP=${OpenMP_CXX_FOUND})")
message(STATUS "  RISC-V 模块: ${YOLOOP_BUILD_RISCV} (march=${YOLOOP_RISCV_ARCH}, abi=${YOLOOP_RISCV_ABI})")
message(STATUS "  A100 后处理: ${YOLOOP_RISCV_A100} (ON reserves one K3 AI core)")
message(STATUS "  CUDA 模块:   ${YOLOOP_BUILD_CUDA} (compiler=${CMAKE_CUDA_COMPILER})")
message(STATUS "  app 模块:    ${YOLOOP_BUILD_APP} (ncnn tag=${NCNN_GIT_TAG}, ncnn Vulkan=${YOLOOP_NCNN_VULKAN})")
message(STATUS "  SpaceMIT ORT: ${YOLOOP_BUILD_SPACEMIT_ORT} (root=${SPACEMIT_ORT_ROOT})")
message(STATUS "===========================================================")
